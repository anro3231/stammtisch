-- Quittieren der Benachrichtigungen durch angemeldete Nutzer erlauben.
-- Die bestehenden RLS-Regeln beschränken Lese-/Schreibzugriff auf eigene Einträge.
GRANT SELECT, INSERT, UPDATE ON TABLE public.fantastix_update_reads TO authenticated;

-- Eigenen Lesezustand aktualisieren, aber keine fremden Einträge.
DROP POLICY IF EXISTS fx_update_reads_update ON public.fantastix_update_reads;
CREATE POLICY fx_update_reads_update ON public.fantastix_update_reads
FOR UPDATE TO authenticated USING (user_id = (SELECT auth.uid()))
WITH CHECK (user_id = (SELECT auth.uid()));
